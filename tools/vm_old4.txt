            int appliedCount = 0;
            for (int i = 0; i < this._allTweaks.Count; i++)
            {
                TweakCardViewModel tw = this._allTweaks[i];
                if (tw.Risk == TweakRisk.Low && tw.Status != TweakStatus.Enabled)
                {
                    tw.IsChecked = true;
                    appliedCount++;
                }
